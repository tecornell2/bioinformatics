#!/usr/bin/env python3
import pandas as pd
import re

# ==== Inputs ====
tbl_file = "turtle_genome.fa.mod.EDTA.intact.fa.cln2.tbl"

# Read genome size from header if you want
genome_size = None
with open(tbl_file) as f:
    for line in f:
        if line.startswith("total length:"):
            m = re.search(r"total length:\s*([\d,]+) bp", line)
            if m:
                genome_size = int(m.group(1).replace(",", ""))
            break
if genome_size is None:
    genome_size = 1.6e9  # fallback

# ==== Parse the file ====
data = []
current_superclass = None

with open(tbl_file) as f:
    for line in f:
        line = line.strip()
        # Skip headers
        if not line or line.startswith(("file name","sequences","total length","GC","bases masked","===","---")):
            continue
        # Match lines with numbers
        m = re.search(r'(\d[\d,]*)\s*bp\s*(\d+\.\d+)?\s*%', line)
        if m:
            length = int(m.group(1).replace(",", ""))
            # Determine the class name: remove colon and leading spaces
            cls = line.split()[0].replace(":", "")
            data.append((cls, length))

# ==== Build DataFrame ====
df = pd.DataFrame(data, columns=["TE_class", "bp_occupied"])
df = df.groupby("TE_class", as_index=False).sum()
df["percent_genome"] = (df["bp_occupied"] / genome_size) * 100
df = df.sort_values("bp_occupied", ascending=False)

# ==== Save + print ====
out_file = "EDTA_repeat_summary.tsv"
df.to_csv(out_file, sep="\t", index=False)
print(df)
print(f"\nSaved summary to {out_file}")

from collections import defaultdict, Counter
import csv

blast_file = "small_scaffolds_blast_results.tsv"
output_summary = "scaffold_taxa_summary_with_flags.tsv"

# Keywords indicating contaminants
contaminant_keywords = [
    "homo sapiens", "bacteria", "fungi", "virus", "microbiome", "microbiota",
    "escherichia", "staphylococcus", "bacillus", "mycoplasma", "saccharomyces",
    "tragelaphus", "mus musculus", "bos taurus", "pan troglodytes", "primates"
]

# Keywords indicating your target organism or relatives
target_keywords = [
    "crotalus", "serpentes", "snake", "ophidia", "viper", "reptile", "boa", "python", "elaphe"
]

# Dictionary to hold scaffold -> list of taxon names
scaffold_hits = defaultdict(list)

with open(blast_file) as f:
    reader = csv.reader(f, delimiter='\t')
    for row in reader:
        if len(row) < 13:
            continue
        scaffold = row[0].strip()
        taxon = row[12].strip().lower()
        scaffold_hits[scaffold].append(taxon)

with open(output_summary, "w") as out:
    out.write("Scaffold\tUnique_Taxa_Count\tTop_Taxa\tContaminant_Flag\tTarget_Flag\n")
    for scaffold, taxa in scaffold_hits.items():
        counts = Counter(taxa)
        top_taxa = ", ".join(f"{name} ({count})" for name, count in counts.most_common(5))

        # Flags
        contaminant_flag = any(any(contam in taxon for contam in contaminant_keywords) for taxon in taxa)
        target_flag = any(any(targ in taxon for targ in target_keywords) for taxon in taxa)

        out.write(f"{scaffold}\t{len(counts)}\t{top_taxa}\t{contaminant_flag}\t{target_flag}\n")

print(f"Flagged summary written to {output_summary}")

input_file = "small_scaffolds_blast_results.tsv"
output_file = "flagged_contaminant_scaffolds.tsv"

seen_scaffolds = set()

with open(input_file, "r") as infile, open(output_file, "w", newline="") as outfile:
    reader = csv.reader(infile, delimiter="\t")
    writer = csv.writer(outfile, delimiter="\t")

    # Write header
    writer.writerow(["Scaffold", "Percent_Identity", "Bit_Score", "Matched_Taxon", "Flagged_As"])

    for row in reader:
        scaffold = row[0]
        pident = float(row[2])
        bitscore = float(row[11])
        stitle = row[12].lower()

        if pident < 95:
            continue  # Skip low identity hits

        # Only flag the first high-confidence hit per scaffold
        if scaffold in seen_scaffolds:
            continue

        # Determine if the hit is squamate or not
        is_squamate = any(keyword in stitle for keyword in SQUAMATE_KEYWORDS)
        flag = "non-squamate" if not is_squamate else "squamate"

        if flag == "non-squamate":
            writer.writerow([scaffold, pident, bitscore, stitle, flag])
            seen_scaffolds.add(scaffold)
